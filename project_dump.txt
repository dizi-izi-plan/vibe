# Деплой за 15–20 минут

## 1. Supabase
1. supabase.com → **New project** (регион ближе к клиентам), задайте пароль БД.
2. **SQL Editor → New query** → вставьте `supabase/schema.sql` → **Run**.
3. Новый запрос → вставьте `supabase/seed.sql` → **Run** (демо-данные для точки `demo`).
4. **Authentication → Users → Add user** → email + пароль владельца (отметьте *Auto Confirm*). Скопируйте UUID пользователя.
5. SQL Editor: `update shops set owner_id = '<UUID>' where id = 'demo';`
6. **Authentication → Providers → Email**: отключите *Allow new users to sign up*, чтобы никто не регистрировался сам.
7. **Project Settings → API**: скопируйте *Project URL* и ключ *anon public*.

## 2. Локальная проверка
```bash
cp .env.example .env     # вставьте URL и anon-ключ
pnpm install
pnpm dev                 # http://localhost:5173/s/demo
```

## 3. Vercel
1. Залейте проект в GitHub. На vercel.com → **Add New → Project** → выберите репозиторий (Framework: Vite, определится сам).
2. **Environment Variables**: `VITE_SUPABASE_URL`, `VITE_SUPABASE_ANON_KEY`.
3. **Deploy**. Сайт будет по адресу `https://<проект>.vercel.app/s/demo`.
4. Для PWA: откройте сайт на телефоне → «Добавить на главный экран». Замените `public/icon-192.png` и `icon-512.png` своими иконками.

## 4. Новая автомастерская
```bash
pnpm new-tenant mojka-ivanova
```
Скрипт создаст `src/tenants/mojka-ivanova/config.ts` и `supabase/tenant_mojka-ivanova.sql`.
Отредактируйте конфиг, выполните SQL в Supabase, привяжите владельца (`update shops set owner_id=…`), сделайте push. Сайт: `/s/mojka-ivanova`.
(Для отдельного домена на точку — добавьте домен в Vercel и редирект на `/s/<shop_id>`.)

## Проверка защиты от овербукинга
В SQL Editor вызовите `select * from create_appointment(...)` дважды на одно время при одном активном боксе: вторая попытка вернёт ошибку `slot_taken`.

import React from 'react'
import ReactDOM from 'react-dom/client'
import { BrowserRouter, Navigate, Route, Routes } from 'react-router-dom'
import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
import './index.css'
import { ShopLayout } from './components/ShopLayout'
import Home from './pages/Home'
import Booking from './pages/Booking'
import MyBooking from './pages/MyBooking'

const qc = new QueryClient({ defaultOptions: { queries: { staleTime: 30_000, retry: 1 } } })

ReactDOM.createRoot(document.getElementById('root')!).render(
  <React.StrictMode>
    <QueryClientProvider client={qc}>
      <BrowserRouter>
        <Routes>
          <Route path="/s/:shopId" element={<ShopLayout />}>
            <Route index element={<Home />} />
            <Route path="book" element={<Booking />} />
            <Route path="my" element={<MyBooking />} />
          </Route>
          <Route path="*" element={<Navigate to="/s/demo" replace />} />
        </Routes>
      </BrowserRouter>
    </QueryClientProvider>
  </React.StrictMode>
)

### ФАЙЛ: src/components/ShopLayout.tsx

import { createContext, useContext, useEffect, type CSSProperties } from 'react'
import { NavLink, Outlet, useParams } from 'react-router-dom'
import { Home, Wrench, CalendarCheck } from 'lucide-react'
import { tenants } from '@/tenants'
import type { TenantConfig } from '@/tenants/types'

const Ctx = createContext<TenantConfig | null>(null)
export const useShop = () => {
  const c = useContext(Ctx)
  if (!c) throw new Error('useShop outside ShopLayout')
  return c
}

export function ShopLayout() {
  const { shopId = '' } = useParams()
  const cfg = tenants[shopId]
  useEffect(() => { if (cfg) document.title = `${cfg.name} — онлайн-запись` }, [cfg])
  if (!cfg) return <div className="p-8 text-white/70">Точка не найдена</div>
  const base = `/s/${cfg.id}`
  const items = [
    { to: base, label: 'Главная', icon: Home, end: true },
    { to: `${base}/book`, label: 'Услуги', icon: Wrench, end: false },
    { to: `${base}/my`, label: 'Моя запись', icon: CalendarCheck, end: false }
  ]
  return (
    <Ctx.Provider value={cfg}>
      <div style={{ '--accent': cfg.accent } as CSSProperties} className="min-h-screen pb-28">
        <Outlet />
        <nav className="fixed inset-x-0 bottom-0 z-40 flex border-t border-white/10 bg-black/50 pb-[env(safe-area-inset-bottom)] backdrop-blur-xl">
          {items.map(({ to, label, icon: Icon, end }) => (
            <NavLink key={to} to={to} end={end}
              className={({ isActive }) => `flex flex-1 flex-col items-center gap-1 py-3 text-xs ${isActive ? 'text-accent' : 'text-white/70'}`}>
              <Icon size={22} />{label}
            </NavLink>
          ))}
        </nav>
      </div>
    </Ctx.Provider>
  )
}

import { useQuery } from '@tanstack/react-query'
import { supabase } from './supabase'
import type { VehicleClass } from '@/tenants/types'

export interface Service {
  id: string
  name: string
  description: string | null
  base_price: number
  base_duration: number
  service_prices_by_class: { vehicle_class: VehicleClass; price: number; duration_min: number }[]
}

export const useServices = (shopId: string) =>
  useQuery({
    queryKey: ['services', shopId],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('services')
        .select('id,name,description,base_price,base_duration,service_prices_by_class(vehicle_class,price,duration_min)')
        .eq('shop_id', shopId)
        .eq('active', true)
        .order('sort')
      if (error) throw error
      return data as unknown as Service[]
    }
  })

// Цена и длительность по классу авто (если для класса не задано — базовые значения)
export const priceFor = (s: Service, c: VehicleClass) => {
  const p = s.service_prices_by_class.find((x) => x.vehicle_class === c)
  return { price: p?.price ?? s.base_price, duration: p?.duration_min ?? s.base_duration }
}

import { useMutation, useQuery } from '@tanstack/react-query'
import { Link, useSearchParams } from 'react-router-dom'
import { formatInTimeZone } from 'date-fns-tz'
import { ru } from 'date-fns/locale'
import { useShop } from '@/components/ShopLayout'
import { supabase } from '@/lib/supabase'

interface Appt { starts_at: string; status: string; service_name: string | null; car_model: string; price: number }
const STATUS: Record<string, string> = { pending: 'Ожидает', in_progress: 'В работе', done: 'Готово', paid: 'Оплачено', cancelled: 'Отменено' }

export default function MyBooking() {
  const cfg = useShop()
  const [sp] = useSearchParams()
  const token = sp.get('t') ?? localStorage.getItem(`booking:${cfg.id}`)
  const q = useQuery({
    queryKey: ['my', token],
    enabled: !!token,
    queryFn: async () => {
      const { data, error } = await supabase.rpc('get_my_appointment', { p_token: token })
      if (error) throw error
      return ((data as Appt[])[0] ?? null)
    }
  })
  const cancel = useMutation({
    mutationFn: async () => {
      const { error } = await supabase.rpc('cancel_appointment', { p_token: token })
      if (error) throw error
    },
    onSuccess: () => q.refetch()
  })
  const a = q.data
  return (
    <div className="mx-auto max-w-2xl space-y-6 px-5 py-8">
      <h1 className="font-display text-4xl font-bold">Моя запись</h1>
      {!a ? (
        <p className="text-white/70">Активных записей нет. <Link className="text-accent underline" to={`/s/${cfg.id}/book`}>Записаться</Link></p>
      ) : (
        <div className="space-y-2 rounded-lg bg-white/10 p-5">
          <div className="text-2xl font-semibold">{a.service_name}</div>
          <div>{formatInTimeZone(new Date(a.starts_at), cfg.timezone, "d MMMM, HH:mm", { locale: ru })}</div>
          <div className="text-white/70">{a.car_model} · {a.price.toLocaleString('ru-RU')} {cfg.currency}</div>
          <div className="font-semibold text-accent">{STATUS[a.status] ?? a.status}</div>
          {a.status === 'pending' && (
            <button onClick={() => cancel.mutate()} disabled={cancel.isPending} className="mt-3 rounded-lg border border-white/30 px-4 py-3">
              Отменить запись
            </button>
          )}
          {cancel.isError && <p className="text-red-400">Отмена недоступна. Позвоните: {cfg.phone}</p>}
        </div>
      )}
    </div>
  )
}

create extension if not exists btree_gist;

create table shops (
  id text primary key,
  name text not null,
  owner_id uuid references auth.users(id) on delete set null
);

create table settings (
  shop_id text primary key references shops(id) on delete cascade,
  timezone text not null default 'Europe/Moscow',
  slot_step_min int not null default 30 check (slot_step_min between 5 and 240),
  -- ключ = ISO-день недели (1=пн … 7=вс), значение = [открытие, закрытие]; нет ключа = выходной
  work_hours jsonb not null default '{"1":["09:00","20:00"],"2":["09:00","20:00"],"3":["09:00","20:00"],"4":["09:00","20:00"],"5":["09:00","20:00"],"6":["10:00","18:00"]}'
);

create table bays (
  id uuid primary key default gen_random_uuid(),
  shop_id text not null references shops(id) on delete cascade,
  name text not null,
  active boolean not null default true
);

create table services (
  id uuid primary key default gen_random_uuid(),
  shop_id text not null references shops(id) on delete cascade,
  name text not null,
  description text,
  base_price int not null check (base_price >= 0),
  base_duration int not null check (base_duration > 0),
  active boolean not null default true,
  sort int not null default 0
);

create table service_prices_by_class (
  service_id uuid not null references services(id) on delete cascade,
  vehicle_class text not null check (vehicle_class in ('sedan','crossover','minivan')),
  price int not null check (price >= 0),
  duration_min int not null check (duration_min > 0),
  primary key (service_id, vehicle_class)
);

create table appointments (
  id uuid primary key default gen_random_uuid(),
  shop_id text not null references shops(id) on delete cascade,
  service_id uuid references services(id) on delete set null,
  bay_id uuid not null references bays(id),
  vehicle_class text not null,
  starts_at timestamptz not null,
  ends_at timestamptz not null,
  status text not null default 'pending' check (status in ('pending','in_progress','done','paid','cancelled')),
  customer_name text not null,
  phone text not null,
  car_model text not null,
  plate text not null,
  price int not null,
  cancel_token uuid not null default gen_random_uuid(),
  created_at timestamptz not null default now(),
  check (ends_at > starts_at),
  -- жёсткая гарантия на уровне БД: в одном боксе записи не пересекаются
  constraint no_bay_overlap exclude using gist (bay_id with =, tstzrange(starts_at, ends_at) with &&) where (status <> 'cancelled')
);
create index on appointments (shop_id, starts_at);
create unique index on appointments (cancel_token);

create table gallery (
  id uuid primary key default gen_random_uuid(),
  shop_id text not null references shops(id) on delete cascade,
  image_path text not null,
  before_path text,
  caption text,
  created_at timestamptz not null default now()
);

-- Свободные слоты на дату (без раскрытия чужих данных)
create or replace function get_free_slots(p_shop_id text, p_service_id uuid, p_class text, p_date date)
returns table (slot timestamptz) language plpgsql security definer stable set search_path = public as $$
declare s settings; dur int; win jsonb; t timestamp; st timestamptz; en timestamptz; nbays int;
begin
  select * into s from settings where shop_id = p_shop_id;
  if not found then return; end if;
  select coalesce(p.duration_min, sv.base_duration) into dur
    from services sv left join service_prices_by_class p on p.service_id = sv.id and p.vehicle_class = p_class
    where sv.id = p_service_id and sv.shop_id = p_shop_id and sv.active;
  if dur is null then return; end if;
  win := s.work_hours -> (extract(isodow from p_date)::int)::text;
  if win is null then return; end if;
  select count(*) into nbays from bays where shop_id = p_shop_id and active;
  t := p_date + (win->>0)::time;
  while t + make_interval(mins => dur) <= p_date + (win->>1)::time loop
    st := t at time zone s.timezone;
    en := st + make_interval(mins => dur);
    if st > now() and (
      select count(distinct a.bay_id) from appointments a
      where a.shop_id = p_shop_id and a.status <> 'cancelled' and tstzrange(a.starts_at, a.ends_at) && tstzrange(st, en)
    ) < nbays then slot := st; return next; end if;
    t := t + make_interval(mins => s.slot_step_min);
  end loop;
end $$;

-- Создание записи: advisory-lock на магазин + поиск свободного бокса + exclusion constraint = нет овербукинга
create or replace function create_appointment(
  p_shop_id text, p_service_id uuid, p_class text, p_starts_at timestamptz,
  p_name text, p_phone text, p_car text, p_plate text)
returns table (appointment_id uuid, token uuid, start_at timestamptz)
language plpgsql security definer set search_path = public as $$
declare s settings; dur int; prc int; fin timestamptz; win jsonb; loc timestamp; bay uuid; rec appointments;
begin
  if length(trim(p_name)) < 2 or length(trim(p_phone)) < 10 or length(trim(p_plate)) < 4 then raise exception 'invalid_input'; end if;
  perform pg_advisory_xact_lock(hashtext(p_shop_id));
  select * into s from settings where shop_id = p_shop_id;
  if not found then raise exception 'shop_not_found'; end if;
  select coalesce(p.duration_min, sv.base_duration), coalesce(p.price, sv.base_price) into dur, prc
    from services sv left join service_prices_by_class p on p.service_id = sv.id and p.vehicle_class = p_class
    where sv.id = p_service_id and sv.shop_id = p_shop_id and sv.active;
  if dur is null then raise exception 'service_not_found'; end if;
  if p_starts_at <= now() then raise exception 'in_past'; end if;
  fin := p_starts_at + make_interval(mins => dur);
  loc := p_starts_at at time zone s.timezone;
  win := s.work_hours -> (extract(isodow from loc)::int)::text;
  if win is null or loc::time < (win->>0)::time or (fin at time zone s.timezone)::time > (win->>1)::time then
    raise exception 'outside_hours';
  end if;
  select b.id into bay from bays b
    where b.shop_id = p_shop_id and b.active
      and not exists (select 1 from appointments a where a.bay_id = b.id and a.status <> 'cancelled'
                      and tstzrange(a.starts_at, a.ends_at) && tstzrange(p_starts_at, fin))
    order by b.name limit 1;
  if bay is null then raise exception 'slot_taken'; end if;
  insert into appointments (shop_id, service_id, bay_id, vehicle_class, starts_at, ends_at, customer_name, phone, car_model, plate, price)
    values (p_shop_id, p_service_id, bay, p_class, p_starts_at, fin, trim(p_name), trim(p_phone), trim(p_car), upper(trim(p_plate)), prc)
    returning * into rec;
  return query select rec.id, rec.cancel_token, rec.starts_at;
exception when exclusion_violation then raise exception 'slot_taken';
end $$;

create or replace function get_my_appointment(p_token uuid)
returns table (starts_at timestamptz, status text, service_name text, car_model text, price int)
language sql security definer stable set search_path = public as
$$ select a.starts_at, a.status, s.name, a.car_model, a.price from appointments a left join services s on s.id = a.service_id where a.cancel_token = p_token $$;

create or replace function cancel_appointment(p_token uuid) returns boolean
language plpgsql security definer set search_path = public as $$
declare n int;
begin
  update appointments set status = 'cancelled' where cancel_token = p_token and status = 'pending' and starts_at > now();
  get diagnostics n = row_count;
  return n > 0;
end $$;

"""Генерация сайтов-тенантов из xlsx: python3 scripts/generate_from_xlsx.py data/services.xlsx"""
import sys, re, json, csv, pathlib
import pandas as pd

src = sys.argv[1] if len(sys.argv) > 1 else 'data/services.xlsx'
df = pd.concat(pd.read_excel(src, sheet_name=None, dtype=str).values(), ignore_index=True).fillna('')
df['key'] = df['2GIS URL'].where(df['2GIS URL'] != '', df['Наименование'] + df['Адрес'])
df = df.drop_duplicates('key').reset_index(drop=True)

TR = dict(zip('абвгдеёжзийклмнопрстуфхцчшщъыьэюя', 'a b v g d e e zh z i y k l m n o p r s t u f kh ts ch sh sch - y - e yu ya'.split()))
def slugify(s):
    t = ''.join(TR.get(c, c) for c in s.lower())
    return re.sub(r'[^a-z0-9]+', '-', t).strip('-') or 'shop'

def phone(raw):
    d = re.sub(r'\D', '', raw.split('(')[0])
    if len(d) == 11 and d[0] in '78': d = '7' + d[1:]
    if len(d) == 10: d = '7' + d
    return f'+7 {d[1:4]} {d[4:7]}-{d[7:9]}-{d[9:11]}' if len(d) == 11 else raw
